// This boundary is a known server limitation: the handler currently rejects
// the 50th wallet before returning a success body. The SDK must surface the 400.
func TestCurrentServerWalletTagBoundary(t *testing.T) {
	gin.SetMode(gin.TestMode)
	r := gin.New()
	r.GET("/v1/tags", api.NewTagsAPIHandler(tagService{}).GetWalletsTags)
	s := httptest.NewServer(r)
	defer s.Close()
	c := cielogo.NewClient("offline-key", cielogo.WithBaseURL(s.URL))
	for _, count := range []int{49, 50} {
		wallets := make([]string, count)
		for i := range wallets {
			wallets[i] = wallet
		}
		_, err := c.GetWalletsTagsV1(context.Background(), &sdk.GetWalletsTagsRequest{Wallets: wallets})
		if count == 49 {
			require.NoError(t, err)
		} else {
			var apiErr *sdkapi.Error
			require.ErrorAs(t, err, &apiErr)
			require.Equal(t, 400, apiErr.StatusCode)
		}
	}
}

func TestSDKAgainstActualHandlers(t *testing.T) {
	gin.SetMode(gin.TestMode)
	r := gin.New()
	r.Use(func(c *gin.Context) {
		ctx := middleware.ContextWithTgHandleId(c.Request.Context(), 17)
		ctx = middleware.ContextWithPlan(ctx, types.PlanApiBuilder)
		c.Request = c.Request.WithContext(ctx)
		c.Next()
	})
	w := api.NewWalletManagementHandler(userRepo{}, backend{t: t})
	r.GET("/v1/tracked-wallets/", w.GetTrackedWallets)
	r.PUT("/v2/tracked-wallets/:wallet", w.UpdateTrackedWalletV2)
	r.POST("/v1/lists/", w.AddList)
	r.GET("/v1/lists/all", w.GetAllLists)
	r.PUT("/v1/lists/:list_id/toggle-follow", w.ToggleFollowList)
	p := api.NewPnLAPIHandler(pnlService{t: t}, tokenService{}, chainRepo{}, false)
	r.GET("/v1/:wallet/pnl/tokens", p.GetTokensPnL)
	r.GET("/v1/:wallet/pnl/nfts", p.GetNftsPnL)
	r.GET("/v1/:wallet/trading-stats", api.NewTradingStatsHandler(tradingService{}).GetTradingStats)
	r.GET("/v1/feed", api.NewFeedAPIHandler(feedService{t: t}, chainRepo{}).GetFeed)
	r.GET("/v1/token/metadata", api.NewTokenApiHandler(tokenService{}).GetTokenMetadata)
	r.GET("/v2/portfolio", api.NewPortfolioAPIHandler(portfolioService{}).GetPortfolioV2)
	server := httptest.NewServer(r)
	defer server.Close()
	c := cielogo.NewClient("offline-key", cielogo.WithBaseURL(server.URL), cielogo.WithPortfolioBaseURL(server.URL))
	ctx := context.Background()
	t.Run("tracked query", func(t *testing.T) {
		resp, err := c.GetTrackedWalletsV1(ctx, &sdk.GetTrackedWalletsRequest{NextObject: sdk.ToRef("2"), ListID: sdk.ToRef(int64(17))})
		require.NoError(t, err)
		require.Equal(t, 3, resp.Pagination.NextObject)
	})
	t.Run("V2 settings", func(t *testing.T) {
		resp, err := c.UpdateTrackedWalletV2(ctx, wallet, &sdk.UpdateTrackedWalletV2Request{Filters: []int{1, 2}, Chains: []int{1}, MinAmountUSD: sdk.ToRef(100.0), NewTrades: sdk.ToRef(false)})
		require.NoError(t, err)
		require.Equal(t, int64(17), resp.ID)
	})
	t.Run("list IDs", func(t *testing.T) {
		resp, err := c.AddWalletsListV1(ctx, &sdk.AddWalletsListRequest{Name: "list", Wallets: []int64{17, 18}})
		require.NoError(t, err)
		require.Equal(t, int64(17), resp.ID)
	})
	t.Run("list filter", func(t *testing.T) {
		resp, err := c.GetAllWalletsListV1(ctx, &sdk.GetAllWalletsListsRequest{FollowOnly: true})
		require.NoError(t, err)
		require.Len(t, resp.List, 1)
	})
	t.Run("toggle envelope", func(t *testing.T) {
		resp, err := c.ToggleFollowWalletsListV1(ctx, 17)
		require.NoError(t, err)
		require.True(t, resp.Followed)
	})
	t.Run("token chain", func(t *testing.T) {
		resp, err := c.GetTokensPnlV1(ctx, &sdk.TokensPnLRequest{Wallet: wallet, Chains: []chains.ChainType{chains.Solana}})
		require.NoError(t, err)
		require.Equal(t, 123.0, resp.Items[0].TotalPnlUSD)
	})
	t.Run("NFT fields", func(t *testing.T) {
		resp, err := c.GetNftsPnlV1(ctx, &sdk.NftsPnLRequest{Wallet: wallet})
		require.NoError(t, err)
		require.Equal(t, "NFT", resp.Items[0].Name)
		require.Equal(t, 123.0, resp.Items[0].ProfitLoss)
	})
	t.Run("trading fields", func(t *testing.T) {
		resp, err := c.GetTradingStatsV1(ctx, &sdk.TradingStatsRequest{Wallet: wallet})
		require.NoError(t, err)
		require.Equal(t, 123.0, resp.PnL)
		require.Equal(t, 0.8, resp.WinRate)
	})
	t.Run("feed filter and swap", func(t *testing.T) {
		resp, err := c.GetFeedV1(ctx, &sdk.FeedRequest{Wallet: wallet, TxTypes: []sdk.TxType{sdk.TxTypeSwap}})
		require.NoError(t, err)
		swap, ok := resp.Items[0].Data.(*sdk.SwapEvent)
		require.True(t, ok)
		require.Equal(t, "SOL", swap.Token0Address)
		require.Equal(t, 2.0, swap.Token0Amount)
	})
	t.Run("unsigned metadata", func(t *testing.T) {
		resp, err := c.GetTokenMetadataV1(ctx, &sdk.TokenMetadataRequest{Chain: sdk.TokenChainEthereum, TokenAddress: wallet})
		require.NoError(t, err)
		require.Equal(t, "Token", resp.Name)
		require.Equal(t, ^uint64(0), resp.Supply)
		require.Equal(t, "18.45", resp.TotalSupplyString)
	})
	t.Run("private portfolio", func(t *testing.T) {
		resp, err := c.GetWalletPortfolioV2(ctx, &sdk.WalletPortfolioV2Request{Wallets: []string{wallet}})
		require.NoError(t, err)
		require.Equal(t, 123.0, resp.TotalUSDValue)
	})
	t.Run("single token", func(t *testing.T) {
		resp, err := c.GetWalletPortfolioV2(ctx, &sdk.WalletPortfolioV2Request{Wallets: []string{"So11111111111111111111111111111111111111112"}, Token: sdk.ToRef("So11111111111111111111111111111111111111112")})
		require.NoError(t, err)
		require.NotNil(t, resp.SingleToken)
		require.Equal(t, 300.0, resp.SingleToken.TotalUSDValue)
	})
}
